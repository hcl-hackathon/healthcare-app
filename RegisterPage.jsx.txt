import React, { useState } from 'react'

function RegisterPage() {
	const [name, setName] = useState('')
	const [email, setEmail] = useState('')
	const [password, setPassword] = useState('')
	const [role, setRole] = useState('')
	const [nameError, setNameError] = useState('')
	const [emailError, setEmailError] = useState('')
	const [passwordError, setPasswordError] = useState('')
	const [roleError, setRoleError] = useState('')

	// Name Validation Function
	const validateName = (name) => {
		const nameRegex = /^[a-zA-Z]+(?:\s[a-zA-Z]+)*$/
		return nameRegex.test(name)
	}

	// Email Validation Function
	const validateEmail = (email) => {
		const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/
		return emailRegex.test(email)
	}

	// Password Validation Function
	const validatePassword = (password) => {
		const passwordRegex = /^(?=.*[a-z])(?=.*[A-Z])(?=.*\d).{6,}$/
		return passwordRegex.test(password)
	}

	const handleSubmit = (event) => {
		event.preventDefault()
		let valid = true

		// Validate Name
		if (!validateName(name)) {
			setNameError('Please enter a valid name (only letters and spaces)')
			valid = false
		} else {
			setNameError('')
		}

		// Validate Email
		if (!validateEmail(email)) {
			setEmailError('Please enter a valid email address')
			valid = false
		} else {
			setEmailError('')
		}

		// Validate Password
		if (!validatePassword(password)) {
			setPasswordError(
				'Password must be at least 6 characters long, with at least one uppercase letter, one lowercase letter, and one number'
			)
			valid = false
		} else {
			setPasswordError('')
		}

		// Validate Role
		if (!role) {
			setRoleError('Please select a role')
			valid = false
		} else {
			setRoleError('')
		}

		// Submit only if valid
		if (valid) {
			console.log('Form Submitted:', { name, email, password, role })
		}
	}

	return (
		<div>
			<div className="register-container">
				<h3>Register Page</h3>
				<form onSubmit={handleSubmit}>
					<div>
						<input
							type="text"
							id="name"
							placeholder="Name"
							value={name}
							onChange={(e) => setName(e.target.value)}
							required
						/>
						{nameError && <p className="error">{nameError}</p>}
					</div>
					<div>
						<input
							type="email"
							id="email"
							placeholder="Email"
							value={email}
							onChange={(e) => setEmail(e.target.value)}
							required
						/>
						{emailError && <p className="error">{emailError}</p>}
					</div>
					<div>
						<input
							type="password"
							id="password"
							placeholder="Password"
							value={password}
							onChange={(e) => setPassword(e.target.value)}
							required
						/>
						{passwordError && (
							<p className="error">{passwordError}</p>
						)}
					</div>
					<div>
						<select
							id="role"
							value={role}
							onChange={(e) => setRole(e.target.value)}
							required
						>
							<option value="">Select Role</option>
							<option value="User">Patient</option>
							<option value="Admin">Provider</option>
						</select>
						{roleError && <p className="error">{roleError}</p>}
					</div>
					<button type="submit">Submit</button>
				</form>
			</div>
		</div>
	)
}

export default RegisterPage
